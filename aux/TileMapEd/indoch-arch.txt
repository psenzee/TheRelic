class MotionHelper
{
    public static float GetAngle(Vector3 a, Vector3 b)
    {
        return Math.Acos(Vector3.Dot(a, b));
    }
    public static float GetAngle(Motion a, Motion m)
    {
        return GetAngle(m.Direction, a.Direction);
    }
    public static Vector3 GetUpAxis(Motion a, Motion m)
    {
        Vector3 up = Vector3.Cross(Vector3.Normalize(m.Direction), Vector3.Normalize(this.Motion.Direction));
        if (!Equals(up.LengthSquared(), 1.0f))
            up = m.Up;
        return up;
    }

class Mover : ICloneable
{
    public Motion Motion    [r]
    public Sphere Bounds    [r]

    public abstract void Update(Motion m)
    
    public Mover(Motion m, Sphere b)
    
    public Mover Clone()
}

class NullMover : Mover
{
    public void Update(Motion m) {}
}

class OrientedMover : Mover
{
    public void Update(Motion m)
    {   
        Quaternion q = null;
        if (!MoverHelper.Collinear(m.Direction, this.Motion.Direction, DELTA))
            q = Quaternion.CreateFromAxisAndAngle(GetUpAxis(m), GetAngle());
        this.Motion = m.Clone();
        if (q != null)
            this.Motion.Orientation = q;
    }
}

{
