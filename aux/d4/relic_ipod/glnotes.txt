class BoolStateSetter
{
public:
    inline BoolStateSetter(int state) : mState(state) {}
    inline void Set(bool value) { if (value) glEnable(mState); else glDisable(mState); }
private:
    int mState;
};

{
public:
    inline BoolClientStateSetter(int state) : mState(state) {}

private:
    int mState;
};

class LightScalarSetter

};

typename <typename SetterT, typename T>
class State
{
public:
    inline State(const SetterT &setter, const T &initial) : mSetter(setter), mState(state) { ForceSet(initial); }
    
    inline void Set(const T &v)             { if (v != mValue) ForceSet(v); }
    inline void ForceSet(const T &v)        { mSetter.Set(value); mValue = v; }
    inline bool Get()                 const { return mValue; }    

private:
    SetterT mSetter;
    T       mState;
    bool    mValue;
};

class GLBoolState            : public State<BoolStateSetter, bool>               { public: inline GLBoolState(int state, bool initial) : State<BoolStateSetter, bool>(BoolStateSetter(state), initial) {} };

class GLLightScalarState     : public State<LightScalarSetter, bool>             { public: inline GLLightScalarState(int light, int state, float initial) : State<LightScalarSetter, float>(LightScalarSetter(light, state), initial) {} };
class GLLightVector4State    : public State<LightVectorSetter<Vector4>, bool>    { public: inline GLLightVector4State(int light, int state, const Vector4 &initial) : State<LightVectorSetter<Vector4>, Vector4>(LightVectorSetter<Vector4>(light, state), initial) {} };
class GLMaterialScalarState  : public State<MaterialScalarSetter, bool>          { public: inline GLMaterialScalarState(int state, float initial) : State<MaterialScalarSetter, float>(MaterialScalarSetter(state), initial) {} };
class GLMaterialVector4State : public State<MaterialVectorSetter<Vector4>, bool> { public: inline GLMaterialVector4State(int state, const Vector4 &initial) : State<MaterialVectorSetter<Vector4>, Vector4>(MaterialVectorSetter<Vector4>(state), initial) {} };
